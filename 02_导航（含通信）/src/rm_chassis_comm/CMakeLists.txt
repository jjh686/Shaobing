cmake_minimum_required(VERSION 3.8)
project(rm_chassis_comm LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)

add_library(chassis_protocol STATIC src/protocol.c src/gimbal_protocol.c firmware/mcu_link.c firmware/can_transport.c
  firmware/basic_framework_port.c
  "方案/底盘通信（USB直连底盘板）/direct_usb.c"
  "方案/底盘通信（USB经云台CAN转发）/gimbal_can.c")
target_include_directories(chassis_protocol PUBLIC include)
target_link_libraries(chassis_protocol PUBLIC m)
add_executable(chassis_node src/chassis_node.cpp)
target_include_directories(chassis_node PRIVATE include)
target_link_libraries(chassis_node chassis_protocol util)
ament_target_dependencies(chassis_node rclcpp rcl_interfaces geometry_msgs std_msgs std_srvs diagnostic_msgs sensor_msgs)
add_executable(sentry_frame_bridge src/sentry_frame_bridge.cpp)
target_include_directories(sentry_frame_bridge PRIVATE include)
ament_target_dependencies(sentry_frame_bridge rclcpp rcl_interfaces geometry_msgs std_msgs sensor_msgs tf2_ros)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(chassis_protocol PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(chassis_node PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  enable_testing()
  foreach(test_name IN ITEMS protocol mcu can port gimbal_protocol)
    add_executable(test_${test_name} test/test_${test_name}.c)
    target_link_libraries(test_${test_name} chassis_protocol)
    add_test(NAME chassis_${test_name} COMMAND test_${test_name})
  endforeach()
  add_executable(test_mux test/test_mux.cpp)
  target_link_libraries(test_mux chassis_protocol)
  add_test(NAME chassis_mux COMMAND test_mux)
  foreach(test_name IN ITEMS frame_math gimbal_clock)
    add_executable(test_${test_name} test/test_${test_name}.cpp)
    target_include_directories(test_${test_name} PRIVATE include)
    add_test(NAME chassis_${test_name} COMMAND test_${test_name})
  endforeach()
  add_executable(test_sentry_ros test/test_sentry_ros.cpp)
  target_include_directories(test_sentry_ros PRIVATE include)
  target_link_libraries(test_sentry_ros chassis_protocol util)
  ament_target_dependencies(test_sentry_ros rclcpp rcl_interfaces geometry_msgs std_msgs std_srvs diagnostic_msgs sensor_msgs tf2_ros tf2_msgs)
  add_test(NAME sentry_ros_pty COMMAND test_sentry_ros)
  set_tests_properties(sentry_ros_pty PROPERTIES TIMEOUT 25)
endif()
install(TARGETS chassis_node sentry_frame_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config 方案 DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md PROTOCOL.md SENTRY_ARCHITECTURE.md VALIDATION.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
